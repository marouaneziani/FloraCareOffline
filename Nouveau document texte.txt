package com.example.floracare.ui.detail

import android.net.Uri
import android.os.Bundle
import android.widget.Toast
import androidx.activity.result.contract.ActivityResultContracts
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import com.bumptech.glide.Glide
import com.example.floracare.data.local.AppDatabase
import com.example.floracare.data.repository.PlantRepository
import com.example.floracare.databinding.ActivityPlantDetailBinding
import com.example.floracare.utils.ImageStorageHelper
import kotlinx.coroutines.launch
import java.io.File

class PlantDetailActivity : AppCompatActivity() {
    private lateinit var b: ActivityPlantDetailBinding
    private lateinit var repo: PlantRepository
    private var plantId: Int? = null
    private var imagePath: String? = null

    private val pickImage = registerForActivityResult(ActivityResultContracts.GetContent()) { uri: Uri? ->
        uri ?: return@registerForActivityResult
        imagePath = ImageStorageHelper.copyImageToPrivateStorage(this, uri)
        imagePath?.let { Glide.with(this).load(File(it)).into(b.imgPreview) }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        b = ActivityPlantDetailBinding.inflate(layoutInflater)
        setContentView(b.root)
        repo = PlantRepository(AppDatabase.get(this).plantDao())
        plantId = if (intent.hasExtra("id")) intent.getIntExtra("id", 0) else null

        plantId?.let { id ->
            lifecycleScope.launch {
                val p = repo.getById(id) ?: return@launch
                b.etName.setText(p.name)
                b.etSpecies.setText(p.species)
                b.etDays.setText(p.wateringIntervalDays.toString())
                imagePath = p.localImageUri
                imagePath?.let { Glide.with(this@PlantDetailActivity).load(File(it)).into(b.imgPreview) }
            }
        }

        b.btnGallery.setOnClickListener { pickImage.launch("image/*") }
        b.btnCamera.setOnClickListener {
            Toast.makeText(this, "Version simple ISTA : utilisez Galerie. Vous pouvez ajouter TakePicture après.", Toast.LENGTH_LONG).show()
        }
        b.btnCancel.setOnClickListener { finish() }
        b.btnSave.setOnClickListener { save() }
    }

    private fun save() {
        val name = b.etName.text.toString().trim()
        val species = b.etSpecies.text.toString().trim()
        val days = b.etDays.text.toString().toIntOrNull()
        if (name.length !in 2..50) { b.etName.error = "Nom entre 2 et 50 caractères"; return }
        if (species.isBlank()) { b.etSpecies.error = "Espèce obligatoire"; return }
        if (days == null || days !in 1..365) { b.etDays.error = "Nombre entre 1 et 365"; return }
        if (plantId == null && imagePath == null) { Toast.makeText(this, "Photo obligatoire", Toast.LENGTH_SHORT).show(); return }
        lifecycleScope.launch {
            repo.savePlant(plantId, name, species, days, imagePath)
            Toast.makeText(this@PlantDetailActivity, "Enregistré", Toast.LENGTH_SHORT).show()
            finish()
        }
    }
}
