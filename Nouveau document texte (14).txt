package com.example.floracare.network

import com.example.floracare.data.models.*
import retrofit2.Response
import retrofit2.http.*

interface ApiService {
    @POST("auth/register")
    suspend fun register(@Body request: RegisterRequest): Response<AuthResponseDto>

    @POST("auth/login")
    suspend fun login(@Body request: LoginRequest): Response<AuthResponseDto>

    @GET("plants")
    suspend fun getPlants(): Response<List<PlantNetworkDto>>
}
