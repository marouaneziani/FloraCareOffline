package com.example.floracare.data.local

import android.database.Cursor
import androidx.room.*
import kotlinx.coroutines.flow.Flow
import com.example.floracare.utils.SyncStatus

@Dao
interface PlantDao {
    @Query("SELECT * FROM plants WHERE syncStatus != 'DELETED' ORDER BY name ASC")
    fun getActivePlants(): Flow<List<PlantEntity>>

    @Query("SELECT * FROM plants WHERE syncStatus != 'DELETED' AND (name LIKE '%' || :q || '%' OR species LIKE '%' || :q || '%') ORDER BY name ASC")
    fun search(q: String): Flow<List<PlantEntity>>

    @Query("SELECT * FROM plants WHERE id = :id LIMIT 1")
    suspend fun getById(id: Int): PlantEntity?

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insert(plant: PlantEntity)

    @Update suspend fun update(plant: PlantEntity)

    @Query("UPDATE plants SET lastWateredTimestamp = :date, syncStatus = CASE WHEN syncStatus='ADDED' THEN 'ADDED' ELSE 'UPDATED' END WHERE id = :id")
    suspend fun waterNow(id: Int, date: Long = System.currentTimeMillis())

    @Query("UPDATE plants SET syncStatus = 'DELETED' WHERE id = :id")
    suspend fun softDelete(id: Int)

    @Query("SELECT * FROM plants WHERE syncStatus != 'NORMAL'")
    suspend fun pendingSync(): List<PlantEntity>

    @Query("UPDATE plants SET syncStatus = 'NORMAL' WHERE id = :id")
    suspend fun markNormal(id: Int)

    @Query("SELECT id, name, species, wateringIntervalDays, lastWateredTimestamp, localImageUri, syncStatus FROM plants WHERE syncStatus != 'DELETED'")
    fun providerQueryAll(): Cursor

    @Query("SELECT id, name, species, wateringIntervalDays, lastWateredTimestamp, localImageUri, syncStatus FROM plants WHERE syncStatus != 'DELETED' AND id = :id")
    fun providerQueryOne(id: Int): Cursor
}
