package com.example.floracare.repository

import android.content.Context
import com.example.floracare.data.datastore.TokenDataStore
import com.example.floracare.data.models.LoginRequest
import com.example.floracare.data.models.RegisterRequest
import com.example.floracare.network.RetrofitClient
import kotlinx.coroutines.flow.first
import retrofit2.HttpException
import java.io.IOException

class AuthRepository(private val context: Context) {
    private val tokenDataStore = TokenDataStore(context)
    private val apiService = RetrofitClient.getApiService(tokenDataStore)

    suspend fun register(username: String, email: String, password: String): Result<Unit> {
        return try {
            val request = RegisterRequest(username, email, password)
            val response = apiService.register(request)

            if (response.isSuccessful) {
                val authResponse = response.body()
                if (authResponse != null && authResponse.token.isNotEmpty()) {
                    tokenDataStore.saveToken(authResponse.token)
                    tokenDataStore.saveUser(authResponse.user.id, authResponse.user.username)
                    Result.success(Unit)
                } else {
                    Result.failure(Exception("Réponse invalide du serveur"))
                }
            } else {
                val errorMessage = response.errorBody()?.string() ?: "Erreur lors de l'inscription"
                Result.failure(Exception(errorMessage))
            }
        } catch (e: IOException) {
            Result.failure(Exception("Erreur réseau : ${e.message}"))
        } catch (e: HttpException) {
            Result.failure(Exception("Erreur serveur : ${e.message}"))
        } catch (e: Exception) {
            Result.failure(Exception("Erreur : ${e.message}"))
        }
    }

    suspend fun login(username: String, password: String): Result<Unit> {
        return try {
            val request = LoginRequest(username, password)
            val response = apiService.login(request)

            if (response.isSuccessful) {
                val authResponse = response.body()
                if (authResponse != null && authResponse.token.isNotEmpty()) {
                    tokenDataStore.saveToken(authResponse.token)
                    tokenDataStore.saveUser(authResponse.user.id, authResponse.user.username)
                    Result.success(Unit)
                } else {
                    Result.failure(Exception("Token invalide reçu du serveur"))
                }
            } else {
                val errorMessage = response.errorBody()?.string() ?: "Email ou mot de passe incorrect"
                Result.failure(Exception(errorMessage))
            }
        } catch (e: IOException) {
            Result.failure(Exception("Erreur réseau : Vérifiez votre connexion"))
        } catch (e: HttpException) {
            Result.failure(Exception("Erreur serveur : ${e.message}"))
        } catch (e: Exception) {
            Result.failure(Exception("Erreur : ${e.message}"))
        }
    }

    suspend fun isLoggedIn(): Boolean {
        return tokenDataStore.isLoggedIn().first()
    }

    suspend fun logout() {
        tokenDataStore.clear()
    }
}
